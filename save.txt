targets_df <- as.data.frame(linkList.max) %>%
  filter(regulatoryGene %in% top_tfs) %>%
  dplyr::select(regulatoryGene, targetGene, weight)

query_genes <- unique(targets_df$targetGene)
bg_genes <- rownames(estrogen_expression)

query_entrez <- mapIds(
  org.Hs.eg.db,
  keys = as.character(query_genes),
  column = "ENTREZID",
  keytype = "SYMBOL",
  multiVals = "first"
)
query_entrez <- na.omit(unique(unname(query_entrez)))

bg_entrez <- mapIds(
  org.Hs.eg.db,
  keys = as.character(bg_genes),
  column = "ENTREZID",
  keytype = "SYMBOL",
  multiVals = "first"
)
bg_entrez <- na.omit(unique(unname(bg_entrez)))


res_list <- list()
for (tf in top_tfs) {
  cat("Running enrichment for:", tf, "\n")
  
  # Extract this TF's targets
  tg <- unique(targets_df$targetGene[targets_df$regulatoryGene == tf])
  
  # Skip if too few targets
  if (length(tg) < 10) {
    message(tf, " skipped (too few targets: ", length(tg), ")")
    next
  }
  
  # Map SYMBOL → ENTREZ
  tg_entrez <- mapIds(
    org.Hs.eg.db,
    keys = as.character(tg),
    column = "ENTREZID",
    keytype = "SYMBOL",
    multiVals = "first"
  )
  tg_entrez <- na.omit(unique(unname(tg_entrez)))
  
  # Run enrichment
  eg <- enrichGO(
    gene          = tg_entrez,
    OrgDb         = org.Hs.eg.db,
    keyType       = "ENTREZID",
    universe      = bg_entrez,
    ont           = "BP",
    pAdjustMethod = "BH",
    pvalueCutoff  = 0.05,
    qvalueCutoff  = 0.2
  )
  
  # Simplify redundant terms
  eg_simpl <- simplify(eg, cutoff = 0.7, by = "p.adjust", select_fun = min)
  
  # Store
  res_list[[tf]] <- eg_simpl
}

for (tf in names(res_list)) {
  write.csv(as.data.frame(res_list[[tf]]),
            file = paste0("ORA_", tf, "_targets.csv"), row.names = FALSE)
}
